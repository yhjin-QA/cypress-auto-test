###############
# SysV Init Information

# description: logserver is the logserver daemon.
### BEGIN INIT INFO
# Provides: logserver
# Required-Start: $network $local_fs $remote_fs

# Default-Start: 2 3 4 5
# Default-Stop: 0 1 6

# 시험에서만 줄일 수 있는 대기 시간(설정 파일 항목이 아님).
STOP_TIMEOUT_SEC=${LOGSERVER_STOP_TIMEOUT_SEC:-40}
[ "$STOP_TIMEOUT_SEC" -ge 1 ] 2>/dev/null || STOP_TIMEOUT_SEC=40

logserver_java_pids() {
    ps -ww -eo pid,args | awk '$2 ~ /(^|\/)java$/ && / -jar [^ ]*logcatch-logserver-[^ /]+[.]jar( |$)/ {print $1}'
}

# 모든 PID 에 같은 상한을 적용한다(여러 JVM 이어도 총 40초). PID 별로 실제 종료를 확인한다.
wait_logserver_shutdown() {
    WAIT_LIMIT=$1
    WAIT_NAME=$2
    shift 2
    WAIT_ELAPSED=0
    while [ "$WAIT_ELAPSED" -lt "$WAIT_LIMIT" ]; do
        WAIT_ALIVE=
        for WAIT_PID in "$@"; do
            if kill -0 "$WAIT_PID" 2>/dev/null; then WAIT_ALIVE=1; fi
        done
        [ -n "$WAIT_ALIVE" ] || return 0
        echo "Waiting for $WAIT_NAME to shutdown ..."
        sleep 1
        WAIT_ELAPSED=$((WAIT_ELAPSED + 1))
    done
    for WAIT_PID in "$@"; do
        if kill -0 "$WAIT_PID" 2>/dev/null; then
            echo "WARNING: $WAIT_NAME (PID $WAIT_PID) did not stop within $WAIT_LIMIT seconds; sending SIGKILL"
            kill -9 "$WAIT_PID" 2>/dev/null || true
        fi
    done
}

case "$1" in
    start)
        process=`ps aux | grep -E "logserver.sh" | grep -v grep | awk '{print $2}'`
        if [ "" != "$process" ] ; then
            echo "logserver.sh is already running or crashed"
        else
            echo "Starting logserver.sh"
        	  sudo -u logcatch -i sh -c '. /etc/profile;nohup /home/logcatch/bin/logserver.sh >> /home/logcatch/proc/logserver/log/logserver-sh.out 2>&1 &'
        fi
        ;;
    stop)
        process=$(ps -ww -eo pid,args | grep -E '(^|[ /])logserver[.]sh( |$)' | grep -v grep | awk '{print $1}')
        # Java 는 루프의 자식이어도 짧은(3초) 강제 종료 대상에서 제외한다. drain 에 최대 40초를 준다.
        LOOP_CHILD_PIDS=
        for LOOP_PID in $process; do
            for CHILD_PID in $(ps -o pid= --ppid "$LOOP_PID"); do
                LOOP_CHILD_PIDS="$LOOP_CHILD_PIDS $CHILD_PID"
            done
        done
