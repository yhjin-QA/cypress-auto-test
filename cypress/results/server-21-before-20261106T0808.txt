#!/bin/sh
#Configurations injected by install_server below....


if [ -f /etc/profile ] ; then
  . /etc/profile
fi

###############
# SysV Init Information
# chkconfig: - 58 74
# description: logserver is the logserver daemon.
### BEGIN INIT INFO
# Provides: logserver
# Required-Start: $network $local_fs $remote_fs
# Required-Stop: $network $local_fs $remote_fs
# Default-Start: 2 3 4 5
# Default-Stop: 0 1 6
# Should-Start: $syslog $named
# Should-Stop: $syslog $named
# Short-Description: start and stop logserver
# Description: logServer daemon
### END INIT INFO

mkdir -p /home/logcatch/conf/
mkdir -p /home/logcatch/proc/logserver/log/

chown -R logcatch:logcatch /home/logcatch/proc/logserver/log/

case "$1" in
    start)
        process=`ps aux | grep -E "logserver.sh" | grep -v grep | awk '{print $2}'`
        if [ "" != "$process" ] ; then
            echo "logserver.sh is already running or crashed"
        else
            echo "Starting logserver.sh"
        	  sudo -u logcatch -i sh -c '. /etc/profile;nohup /home/logcatch/bin/logserver.sh >> /home/logcatch/proc/logserver/log/logserver-sh.out 2>&1 &'
        fi
        ;;
    stop)
        process=`ps aux | grep -E "logserver.sh" | grep -v grep | awk '{print $2}'`
	if [ -z $process ] ; then
            echo "logserver.sh is not running"
        else
            echo "logserver.sh shutdown."
			kill -9 $(ps -o pid= --ppid ${process});kill -9 ${process}
            while [ -x /proc/${process} ]
            do
                echo "Waiting for logserver.sh to shutdown ..."
                sleep 1
            done
        fi

        process=`ps aux | grep -E "logcatch-logserver" | grep -v grep | awk '{print $2}'`
	if [ -z $process ] ; then
            echo "logcatch-logserver is not running"
        else
            echo "logcatch-logserver shutdown."
            kill -9 ${process}
            while [ -x /proc/${process} ]
            do
                echo "Waiting for logcatch-logserver to shutdown ..."
                sleep 1
            done
        fi
        ;;
    restart)
        $0 stop
        $0 start
        ;;
    *)
        echo "Please use start, stop, restart or status as first argument"
        ;;
esac
